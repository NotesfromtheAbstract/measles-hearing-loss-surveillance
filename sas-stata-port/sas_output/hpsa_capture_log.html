 123        /******************************************************************************
 124          Step 3. Round to three decimals the way R's round(x, 3) does.
 125          Pick the nearer thousandth. When the value sits exactly halfway
 126          between two thousandths, keep the even thousandth.
 127          Gaines County is a halfway case: 393 * 0.0085 = 3.3405, and the
 128          printed raw hearing-loss count is 3.340.
 129          The SAS ROUND function uses the other halfway rule and prints 3.341
 130          for that row, so this step does not call ROUND.
 131        ******************************************************************************/
 132        data rounded;
 133          set modeled;
 134          array xin{3} capture_prob expected_hl_raw expected_hl_captured;
 135          array thou{3} thou_cap thou_raw thou_capd;
 136          do i = 1 to 3;
 137            x = xin{i};
 138            x10 = x * 1000;
 139            i10 = floor(x10);
 140            xd = i10 / 1000;
 141            xu = ceil(x10) / 1000;
 142            du = xu - x;
 143            dd = x - xd;
 144            if du < dd or (mod(i10, 2) = 1 and du = dd) then thou{i} = ceil(x10);
 145            else thou{i} = i10;
 146          end;
 147          drop i x x10 i10 xd xu du dd;
 148        run;
